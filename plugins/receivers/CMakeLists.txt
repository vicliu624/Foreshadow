function(amc_receiver_module target factory priority)
  add_library(${target} MODULE ${ARGN} common/entry.cpp)
  set_property(GLOBAL APPEND PROPERTY AMC_RECEIVER_MODULE_TARGETS ${target})
  target_link_libraries(${target} PRIVATE signalscope_receiver_sdk)
  target_compile_definitions(${target} PRIVATE AMC_RECEIVER_FACTORY=${factory} AMC_RECEIVER_PRIORITY=${priority}
      AMC_RECEIVER_CATALOG="${target}.i18n")
  set_target_properties(${target} PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
      LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/receiver-plugins$<0:>"
      RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/receiver-plugins$<0:>")
  set(receiver_resource_compatibility "")
  if(APPLE)
    set(receiver_install_dir Foreshadow.app/Contents/MacOS/receiver-plugins)
    set(receiver_resource_dir Foreshadow.app/Contents/Resources/signalscope/receivers)
    set(receiver_resource_compatibility ",\"application\":{\"minimum\":\"0.2.6\",\"maximum\":\"0.3.0\"}")
    target_compile_definitions(${target} PRIVATE AMC_RECEIVER_CATALOG_ROOT="../../Resources/signalscope/receivers")
  elseif(WIN32)
    set(receiver_install_dir bin/receiver-plugins)
  else()
    set(receiver_install_dir lib/signalscope/receivers)
  endif()
  if(NOT APPLE)
    set(receiver_resource_dir "${receiver_install_dir}")
  endif()
  install(TARGETS ${target} LIBRARY DESTINATION ${receiver_install_dir} COMPONENT SignalScope
      RUNTIME DESTINATION ${receiver_install_dir} COMPONENT SignalScope)
  string(REPLACE "amc-receiver-" "" receiver_id "${target}")
  if(receiver_id STREQUAL "rtlsdr")
    set(receiver_directory rtl_sdr)
  else()
    set(receiver_directory "${receiver_id}")
  endif()
  set(catalog_names "")
  foreach(domain IN ITEMS common "${receiver_directory}")
    if(domain STREQUAL "common")
      set(catalog_dir amc-receiver-common.i18n)
    else()
      set(catalog_dir "${target}.i18n")
    endif()
    list(APPEND catalog_names "\"${catalog_dir}\"")
    file(GLOB catalogs CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${domain}/i18n/*.json")
    set_property(TARGET ${target} APPEND PROPERTY AMC_RECEIVER_CATALOG_DIRECTORIES
        "${PROJECT_BINARY_DIR}/receiver-plugins/${catalog_dir}")
    foreach(catalog IN LISTS catalogs)
      get_filename_component(locale "${catalog}" NAME)
      configure_file("${catalog}" "${PROJECT_BINARY_DIR}/receiver-plugins/${catalog_dir}/${locale}" COPYONLY)
    endforeach()
    install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/${domain}/i18n/"
        DESTINATION "${receiver_resource_dir}/${catalog_dir}" COMPONENT SignalScope)
  endforeach()
  string(JOIN "," catalog_names_json ${catalog_names})
  set(declaration "${PROJECT_BINARY_DIR}/receiver-plugins/${target}.resources.json")
  file(GENERATE OUTPUT "${declaration}" CONTENT "{\"schema\":1,\"catalogs\":[${catalog_names_json}]${receiver_resource_compatibility}}\n")
  set_property(TARGET ${target} PROPERTY AMC_RECEIVER_RESOURCE_DECLARATION "${declaration}")
  install(FILES "${declaration}" DESTINATION "${receiver_resource_dir}" COMPONENT SignalScope)
endfunction()
function(amc_receiver_windows_runtime target)
  set(runtime_files "")
  foreach(runtime IN LISTS ARGN)
    list(APPEND runtime_files "$<TARGET_FILE:${runtime}>")
    install(FILES "$<TARGET_FILE:${runtime}>" DESTINATION bin/receiver-plugins COMPONENT SignalScope)
  endforeach()
  add_custom_command(TARGET ${target} POST_BUILD
      COMMAND "${CMAKE_COMMAND}" "-DAMC_RECEIVER_DIR=$<TARGET_FILE_DIR:${target}>"
          "-DAMC_RECEIVER_NAME=${target}" "-DAMC_RECEIVER_RUNTIME_FILES=${runtime_files}"
          -P "${PROJECT_SOURCE_DIR}/cmake/CopyReceiverRuntime.cmake" VERBATIM)
  install(FILES "${PROJECT_BINARY_DIR}/receiver-plugins/${target}.runtime.json"
      DESTINATION bin/receiver-plugins COMPONENT SignalScope)
endfunction()

option(SIGNALSCOPE_RTLSDR "Enable the librtlsdr receiver plugin" ON)
option(SIGNALSCOPE_REQUIRE_RTLSDR "Require librtlsdr" OFF)
set(rtlsdr_link "")
if(SIGNALSCOPE_RTLSDR)
  if(WIN32)
    find_package(rtlsdr CONFIG QUIET)
    if(TARGET rtlsdr::rtlsdr_static)
      set(rtlsdr_link rtlsdr::rtlsdr_static)
    endif()
  endif()
  if(NOT rtlsdr_link)
    find_path(RTLSDR_INCLUDE_DIR rtl-sdr.h)
    find_library(RTLSDR_LIBRARY NAMES rtlsdr)
    if(RTLSDR_INCLUDE_DIR AND RTLSDR_LIBRARY)
      target_include_directories(amc-receiver-rtlsdr PRIVATE ${RTLSDR_INCLUDE_DIR})
      set(rtlsdr_link ${RTLSDR_LIBRARY})
    endif()
  endif()
endif()
if(rtlsdr_link)
  target_compile_definitions(amc-receiver-rtlsdr PRIVATE SIGNALSCOPE_HAS_RTLSDR=1)
  target_link_libraries(amc-receiver-rtlsdr PRIVATE ${rtlsdr_link})
elseif(SIGNALSCOPE_REQUIRE_RTLSDR)
  message(FATAL_ERROR "The RTL-SDR plugin requires librtlsdr")
endif()

if(SIGNALSCOPE_SDRPLAY)
  find_package(SoapySDR 0.8.1 QUIET CONFIG)
  if(TARGET SoapySDR)
    add_library(signalscope_soapy_driver INTERFACE)
    target_link_libraries(signalscope_soapy_driver INTERFACE SoapySDR)
    target_link_libraries(amc-receiver-sdrplay PRIVATE signalscope_soapy_driver)
    target_compile_definitions(amc-receiver-sdrplay PRIVATE SIGNALSCOPE_HAS_SOAPY=1)
    include("${PROJECT_SOURCE_DIR}/cmake/RemoteReceiver.cmake")
    get_target_property(soapy_type SoapySDR TYPE)
    if(WIN32 AND soapy_type STREQUAL "SHARED_LIBRARY")
      amc_receiver_windows_runtime(amc-receiver-sdrplay SoapySDR)
    endif()
  endif()
endif()

endif()
