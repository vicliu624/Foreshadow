cmake_minimum_required(VERSION 3.20)
# This project deliberately does not add_subdirectory() or link the AMC host.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(AmcExampleReceiver LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  find_path(AMC_RECEIVER_SDK_INCLUDE signalscope/adapters/receivers/plugin_api.h REQUIRED)
  find_path(AMC_JSON_INCLUDE_DIR nlohmann/json.hpp)
  if(AMC_JSON_INCLUDE_DIR)
    add_library(amc_example_json INTERFACE)
    target_include_directories(amc_example_json INTERFACE "${AMC_JSON_INCLUDE_DIR}")
    add_library(nlohmann_json::nlohmann_json ALIAS amc_example_json)
  else()
    find_package(nlohmann_json 3.12 REQUIRED CONFIG)
  endif()
endif()
add_library(amc-receiver-test MODULE test_receiver.cpp)
target_include_directories(amc-receiver-test PRIVATE "${AMC_RECEIVER_SDK_INCLUDE}")
target_link_libraries(amc-receiver-test PRIVATE nlohmann_json::nlohmann_json)
set_target_properties(amc-receiver-test PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
file(GLOB receiver_catalogs CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/i18n/*.json")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(receiver_catalog_output "${CMAKE_CURRENT_BINARY_DIR}/amc-receiver-test.i18n")
else()
  set(receiver_catalog_output "${PROJECT_BINARY_DIR}/test-receiver-plugins/amc-receiver-test.i18n")
endif()
foreach(catalog IN LISTS receiver_catalogs)
  get_filename_component(locale "${catalog}" NAME)
  configure_file("${catalog}" "${receiver_catalog_output}/${locale}" COPYONLY)
endforeach()
# No install rule: simulated receivers are test fixtures, not production devices.
