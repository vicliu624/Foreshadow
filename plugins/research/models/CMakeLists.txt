target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/registry.cpp")
foreach(model IN ITEMS tone am fm fsk chirp lora psk ofdm)
    target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/${model}/provider.cpp")
    file(GLOB catalogs CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${model}/i18n/*.json")
    foreach(catalog IN LISTS catalogs)
        get_filename_component(locale "${catalog}" NAME)
        configure_file("${catalog}" "${PROJECT_BINARY_DIR}/resources/reference-${model}/i18n/${locale}" COPYONLY)
    endforeach()
    install(DIRECTORY "${model}/i18n/" DESTINATION "share/signalscope/reference-${model}/i18n" COMPONENT SignalScope)
    if(APPLE)
        install(DIRECTORY "${model}/i18n/" DESTINATION "Foreshadow.app/Contents/Resources/signalscope/reference-${model}/i18n" COMPONENT SignalScope)
    endif()
endforeach()
