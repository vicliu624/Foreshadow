# Compose this provider's catalogs from its shared PHY resources and its own
# overrides. Locale discovery and per-key fallback remain resource driven.
file(GLOB paired_catalogs CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/lora/paired_i18n/*.json")
set(paired_locales)
foreach(catalog IN LISTS lora_catalogs paired_catalogs)
    get_filename_component(locale "${catalog}" NAME)
    list(APPEND paired_locales "${locale}")
endforeach()
list(REMOVE_DUPLICATES paired_locales)
foreach(locale IN LISTS paired_locales)
    set(merged "{}")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/lora/i18n/${locale}")
        file(READ "${CMAKE_CURRENT_SOURCE_DIR}/lora/i18n/${locale}" merged)
        foreach(identity label about)
            string(JSON identity_type ERROR_VARIABLE identity_error TYPE "${merged}" "${identity}")
            if(NOT identity_error)
                string(JSON merged REMOVE "${merged}" "${identity}")
            endif()
        endforeach()
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/lora/paired_i18n/${locale}")
        file(READ "${CMAKE_CURRENT_SOURCE_DIR}/lora/paired_i18n/${locale}" own)
        string(JSON count LENGTH "${own}")
        if(count GREATER 0)
            math(EXPR last "${count}-1")
            foreach(index RANGE ${last})
                string(JSON key MEMBER "${own}" ${index})
                string(JSON value GET "${own}" "${key}")
                string(REPLACE "\\" "\\\\" value "${value}")
                string(REPLACE "\"" "\\\"" value "${value}")
                string(REPLACE "\n" "\\n" value "${value}")
                string(REPLACE "\r" "\\r" value "${value}")
                string(REPLACE "\t" "\\t" value "${value}")
                string(JSON merged SET "${merged}" "${key}" "\"${value}\"")
            endforeach()
        endif()
    endif()
    set(destination "${PROJECT_BINARY_DIR}/resources/recovery-lora-paired/i18n/${locale}")
    file(WRITE "${destination}" "${merged}\n")
    install(FILES "${destination}" DESTINATION share/signalscope/recovery-lora-paired/i18n COMPONENT SignalScope)
    if(APPLE)
        install(FILES "${destination}" DESTINATION Foreshadow.app/Contents/Resources/signalscope/recovery-lora-paired/i18n COMPONENT SignalScope)
    endif()
endforeach()
