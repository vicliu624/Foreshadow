# This target owns upstream coding symbols; no generic consumer imports them.
amc_compliance_declaration("${CMAKE_CURRENT_SOURCE_DIR}/compliance-wspr.json")
add_library(foreshadow_wspr_fec STATIC vendor/wsjtx/fano.c vendor/wsjtx/tab.c)
target_compile_definitions(foreshadow_wspr_fec PRIVATE
    fano=foreshadow_wspr_fano encode=foreshadow_wspr_encode Partab=foreshadow_wspr_parity)
set_target_properties(foreshadow_wspr_fec PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/core/fec.cpp")
target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/core/message.cpp")
target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/core/channel.cpp")
target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/core/search.cpp")
target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/core/window.cpp")
target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/live_input.cpp")
target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/downconverter.cpp")
target_sources(signalscope_dsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/recording.cpp")
target_link_libraries(signalscope_dsp PRIVATE foreshadow_wspr_fec)
file(GLOB catalogs CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/i18n/*.json")
foreach(catalog IN LISTS catalogs)
    get_filename_component(locale "${catalog}" NAME)
    configure_file("${catalog}" "${PROJECT_BINARY_DIR}/resources/wspr/i18n/${locale}" COPYONLY)
endforeach()
install(DIRECTORY i18n/ DESTINATION share/signalscope/wspr/i18n COMPONENT SignalScope)
if(APPLE)
    install(DIRECTORY i18n/ DESTINATION Foreshadow.app/Contents/Resources/signalscope/wspr/i18n COMPONENT SignalScope)
endif()
install(FILES vendor/wsjtx/COPYING vendor/wsjtx/origin.json DESTINATION share/licenses/ams/wspr-wsjtx COMPONENT SignalScope)
if(APPLE)
    install(FILES vendor/wsjtx/COPYING vendor/wsjtx/origin.json DESTINATION Foreshadow.app/Contents/Resources/licenses/wspr-wsjtx COMPONENT SignalScope)
endif()
