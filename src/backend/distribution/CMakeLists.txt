string(TOLOWER "${CMAKE_SYSTEM_NAME}" distribution_os)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" distribution_arch)
if(distribution_arch MATCHES "^(amd64|x86_64)$")
  set(distribution_arch x86_64)
elseif(distribution_arch MATCHES "^(arm64|aarch64)$")
  set(distribution_arch arm64)
endif()
if(WIN32)
  set(distribution_os windows)
elseif(APPLE)
  set(distribution_os macos)
elseif(EXISTS /etc/os-release)
  file(STRINGS /etc/os-release os_id REGEX "^ID=")
  file(STRINGS /etc/os-release os_version REGEX "^VERSION_ID=")
  string(REGEX REPLACE "^ID=\"?([^\"]+)\"?$" "\\1" os_id "${os_id}")
  string(REGEX REPLACE "^VERSION_ID=\"?([^\"]+)\"?$" "\\1" os_version "${os_version}")
  set(distribution_os "linux-${os_id}${os_version}")
endif()
set(AMC_DISTRIBUTION_PLATFORM "${distribution_os}-${distribution_arch}" CACHE STRING "Distribution ABI/platform baseline")
file(READ "${PROJECT_SOURCE_DIR}/resources/distribution/trust.json" AMC_DISTRIBUTION_TRUST)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/resources/distribution/trust.json")
configure_file(config.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/distribution_config.hpp" @ONLY)
configure_file("${PROJECT_SOURCE_DIR}/resources/distribution/bundled.json.in"
    "${PROJECT_BINARY_DIR}/resources/distribution/bundled.json" @ONLY)
file(GLOB distribution_catalogs CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/resources/distribution/i18n/*.json")
foreach(catalog IN LISTS distribution_catalogs)
  get_filename_component(locale "${catalog}" NAME)
  configure_file("${catalog}" "${PROJECT_BINARY_DIR}/resources/distribution/i18n/${locale}" COPYONLY)
endforeach()
install(DIRECTORY "${PROJECT_BINARY_DIR}/resources/distribution/" DESTINATION share/signalscope/distribution COMPONENT SignalScope)
if(APPLE)
  install(DIRECTORY "${PROJECT_BINARY_DIR}/resources/distribution/" DESTINATION Foreshadow.app/Contents/Resources/signalscope/distribution COMPONENT SignalScope)
endif()
add_library(signalscope_distribution STATIC catalog.cpp store.cpp service.cpp runtime.cpp package_types.cpp)
target_include_directories(signalscope_distribution PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(signalscope_distribution PUBLIC signalscope_iq PRIVATE OpenSSL::Crypto LibArchive::LibArchive)
if(NOT MSVC)
  target_compile_options(signalscope_distribution PRIVATE -Wall -Wextra -Wpedantic)
endif()
