add_library(signalscope_transport mpeg_ts.cpp broadcast_text.cpp)
target_link_libraries(signalscope_transport PUBLIC signalscope_iq)
file(GLOB catalogs CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/i18n/*.json")
foreach(catalog IN LISTS catalogs)
  get_filename_component(locale "${catalog}" NAME)
  configure_file("${catalog}" "${PROJECT_BINARY_DIR}/resources/broadcast-text/i18n/${locale}" COPYONLY)
endforeach()
install(DIRECTORY i18n/ DESTINATION share/signalscope/broadcast-text/i18n COMPONENT SignalScope)
if(APPLE)
  install(DIRECTORY i18n/ DESTINATION Foreshadow.app/Contents/Resources/signalscope/broadcast-text/i18n COMPONENT SignalScope)
endif()

  find_package(Iconv REQUIRED)
  target_link_libraries(signalscope_transport PRIVATE Iconv::Iconv)
endif()
