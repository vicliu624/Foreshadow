cmake_minimum_required(VERSION 3.20)
project(Foreshadow VERSION 0.2.16 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
include(FetchContent)
include(cmake/DesktopLanguageResources.cmake)
include(cmake/MacRuntime.cmake)
include(cmake/Compliance.cmake)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(json)
set(KISSFFT_TEST OFF CACHE BOOL "" FORCE)
set(KISSFFT_TOOLS OFF CACHE BOOL "" FORCE)
set(KISSFFT_STATIC ON CACHE BOOL "" FORCE)
set(KISSFFT_PKGCONFIG OFF CACHE BOOL "" FORCE)
FetchContent_Declare(kissfft URL https://github.com/mborgerding/kissfft/archive/refs/tags/131.1.0.tar.gz
  URL_HASH SHA256=76c1aac87ddb7258f34b08a13f0eebf9e53afa299857568346aa5c82bcafaf1a)
FetchContent_MakeAvailable(kissfft)
find_package(Threads REQUIRED)
add_subdirectory(src/backend/iq)
add_subdirectory(src/backend/transport)
add_subdirectory(src/backend/media)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(LibArchive REQUIRED)
add_subdirectory(src/backend/distribution)
add_subdirectory(src/adapters/receivers)
add_subdirectory(plugins/receivers)
add_subdirectory(src/adapters/modulation)
add_subdirectory(src/backend)
if(MSVC)
  target_compile_options(signalscope_iq PUBLIC /utf-8 /W4)
else()
  foreach(target IN ITEMS signalscope_iq signalscope_receivers signalscope_modulation signalscope_dsp signalscope_core)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
endif()
add_subdirectory(src/frontend/cli)
add_subdirectory(plugins/external/power_report)
option(SIGNALSCOPE_GUI "Build the Qt desktop application" ON)
option(SIGNALSCOPE_REQUIRE_GUI "Fail configuration if desktop dependencies are unavailable" OFF)
option(SIGNALSCOPE_REQUIRE_AUDIO "Require Qt Multimedia desktop playback" OFF)
if((SIGNALSCOPE_REQUIRE_GUI OR SIGNALSCOPE_REQUIRE_AUDIO) AND NOT SIGNALSCOPE_GUI)
  message(FATAL_ERROR "Desktop or audio requirements cannot be used with SIGNALSCOPE_GUI=OFF")
endif()
if(SIGNALSCOPE_GUI)
  set(desktop_qt_minimum 6.2)
  if(APPLE)
    set(desktop_qt_minimum 6.8)
  endif()
  if(SIGNALSCOPE_REQUIRE_GUI OR SIGNALSCOPE_REQUIRE_AUDIO)
    find_package(Qt6 ${desktop_qt_minimum} REQUIRED COMPONENTS Widgets Network LinguistTools)
  else()
    find_package(Qt6 ${desktop_qt_minimum} QUIET COMPONENTS Widgets Network LinguistTools)
  endif()
  if(Qt6_FOUND)
    set(desktop_enabled ON)
  else()
    message(STATUS "Qt6 Widgets or LinguistTools not found; CLI remains available")
  endif()
endif()
include(cmake/ModuleLinkRules.cmake)
include(CTest)
option(SIGNALSCOPE_LEGACY_TESTS "Enable tests that still require Python" OFF)
if(BUILD_TESTING)
  if(NOT TARGET Qt6::Core)
    find_package(Qt6 6.2 QUIET COMPONENTS Core)
  endif()
endif()
if(TARGET Qt6::Core)
  include(cmake/DtmbReceiver.cmake)
endif()
option(SIGNALSCOPE_TELEVISION "Build the native desktop television receiver and media decoder" ON)
if(desktop_enabled AND SIGNALSCOPE_TELEVISION)
  include(cmake/MediaDecoder.cmake)
endif()
if(BUILD_TESTING OR desktop_enabled)
  add_subdirectory(tools)
endif()
if(desktop_enabled)
  add_subdirectory(src/adapters/distribution)
  add_subdirectory(src/frontend/desktop)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
